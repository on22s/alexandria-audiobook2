# On-demand mutation testing (tools/mutation_check.sh, .github/workflows/mutation.yml).
#
# NOT installed by app/requirements.txt: that file is the Pinokio runtime install,
# and mutation testing is never part of the required CI path.
#
# mutmut 2.x on purpose. mutmut 3+ runs pytest only; this project's suite is
# unittest (GUIDE.md: run `python -m unittest tests.test_x` from app/), and 2.x
# is the line that accepts an arbitrary `runner` command in [mutmut].
#
# Install into the interpreter used for the check, e.g. app/env:
#   app/env/bin/python -m pip install -r app/requirements-mutation.txt
#
# mutmut 2.4.5 itself declares Python >=3.7, but the pinned click==8.4.1 and
# transitive pony>=0.7.20 require Python >=3.10 (CI and app/env are 3.10.20).
# All dependencies are pure Python (click, parso, pony, toml, glob2, junit-xml).
mutmut==2.4.5

# Pin the exact click version tested with mutmut 2.4.5 so installing this file
# cannot silently move a shared environment to an incompatible click.
click==8.4.1
