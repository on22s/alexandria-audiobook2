"""Check run completeness and summarize paired held-out speaker results."""
import hashlib
import json
from pathlib import Path
import random
import statistics

ROOT = Path(__file__).resolve().parent
RUN = ROOT / "run2"


def load(name):
    return json.loads((RUN / name).read_text())


def main():
    protocol, results = load("protocol.json"), load("results.json")
    if results["status"] != "complete":
        raise ValueError("Run did not complete")
    if results["protocol_sha256"] != hashlib.sha256((RUN / "protocol.json").read_bytes()).hexdigest():
        raise ValueError("Protocol identity mismatch")
    if results["script_sha256"] != hashlib.sha256((ROOT / "run_experiment.py").read_bytes()).hexdigest():
        raise ValueError("Runner changed after measurement")
    sets = {split: {x["speaker"] for x in protocol["speakers"][split]}
            for split in ("calibration", "test")}
    if sets["calibration"] & sets["test"]:
        raise ValueError("Calibration and test speakers overlap")
    for split in sets:
        rows = load(split + "_rows.json")
        expected = {(s, duration, arm) for s in sets[split]
                    for duration in protocol["durations_seconds"] for arm in protocol["arms"]}
        observed = {(r["speaker"], r["duration"], r["arm"]) for r in rows}
        if observed != expected or len(rows) != len(expected):
            raise ValueError("Missing or duplicate measurement cells")
        if any(r["status"] != "measured" for r in rows):
            raise ValueError("Required measurement unavailable")
        if any(r["donor"] not in sets[split] for r in rows):
            raise ValueError("Donor crossed split")
    if any(r["status"] != "not_measured" for r in load("availability_controls.json")):
        raise ValueError("Silence/short-clip controls did not refuse scoring")
    test_rows = load("test_rows.json")
    thresholds = load("frozen_thresholds.json")
    per_speaker = []
    for speaker in sorted(sets["test"]):
        primary = [r for r in test_rows if r["speaker"] == speaker and r["arm"] == "swap_4s"]
        deltas = [int(r["window_min"] < thresholds["window_min"]["threshold"]) -
                  int(r["whole"] < thresholds["whole"]["threshold"]) for r in primary]
        per_speaker.append({"speaker": speaker, "primary_recall_difference": statistics.mean(deltas)})
    rng = random.Random(20261023)
    deltas = [r["primary_recall_difference"] for r in per_speaker]
    boot = sorted(statistics.mean(rng.choices(deltas, k=len(deltas))) for _ in range(10000))
    analysis = {"checks": "passed", "required_measured_cells": 192,
                "per_test_speaker": per_speaker,
                "primary_recall_difference": statistics.mean(deltas),
                "speaker_cluster_bootstrap95": [boot[250], boot[9749]],
                "note": "Small controlled sample; bootstrap is exploratory, not evidence of generated-audio performance."}
    (RUN / "analysis.json").write_text(json.dumps(analysis, indent=2) + "\n")
    scores = results["summaries"]["test"]["all"]
    table = ["| Held-out condition | Whole-clip flags | Window-min flags |",
             "| --- | ---: | ---: |"]
    for arm in protocol["arms"]:
        a, b = scores["whole"][arm], scores["window_min"][arm]
        table.append(f"| {arm} | {a['flagged']}/{a['total']} | {b['flagged']}/{b['total']} |")
    document = f"""# CPU speaker-window experiment

Completed using CPU-only ECAPA with two threads and process niceness 15.
Elapsed time: {results['seconds']:.1f} seconds. No live application code or GPU
job was modified. Existing cached model weights were used offline.

The corpus is [AISHELL-3](https://www.openslr.org/93/), supplied by Beijing Shell
Shell Technology Co., Ltd., under Apache-2.0. Calibration and test each use eight
different speakers, balanced by gender, with same-gender donor voices.
Reference, target, and replacement audio come from separate source utterances.
Each target lasts 24 or 48 seconds. Windows are eight seconds with four-second
stride. Raw source hashes are recorded in `run2/inputs.json`.

Thresholds were frozen after calibration and before test scoring. Each metric
maximizes four-second swap detection subject to at most 5% calibration control
false alarms. Whole-clip threshold: {thresholds['whole']['threshold']:.6f};
window-min threshold: {thresholds['window_min']['threshold']:.6f}.

{chr(10).join(table)}

Wrong-speaker swaps are positive controls. Clean, same-speaker splice, and
local gain-change clips are negative controls: a flag is a false alarm there.
Combined held-out window-min false-alarm rate:
{results['test_window_control_false_positive_rate']:.1%}.
Paired four-second detection difference (window minus whole):
{analysis['primary_recall_difference']:+.1%}; exploratory speaker-cluster bootstrap
95% interval [{boot[250]:+.1%}, {boot[9749]:+.1%}].

Predeclared pilot decision: **{'GO' if results['go'] else 'NO-GO'}**. This tests
whether window scoring deserves further investigation, not whether a production
threshold or automatic regeneration should ship.

All 192 required cells were measured. Silence and shorter-than-window controls
were explicitly unmeasured. Completion, split separation, unique cells, donor
separation, and protocol/runner hashes passed independent artifact checks.
An earlier attempt (`run1`) stopped before scoring because a local speaker had
insufficient audio; the failure is retained. The rerun preflights corpus duration.

Limits: one Mandarin human-speech corpus, eight held-out speakers, synthetic
concatenations/splices, and correlated clips. Signal coverage is an energy proxy,
not validated speech activity. This is windowed ECAPA, not the paper's WavLM
wSIM. It has not measured real TTS voice changes, emotion changes, multilingual
robustness, ASR quality, LACI recovery, or pronunciation-adapter training.

Artifacts: `run2/protocol.json`, `run2/model_identity.json`,
`run2/frozen_thresholds.json`, `run2/calibration_rows.json`, `run2/test_rows.json`,
`run2/results.json`, and `run2/analysis.json`. Runners are `run_experiment.py`
and `analyze_results.py`. The runner accepts a fresh output-directory argument;
the analyzer targets the preserved `run2` results.
"""
    (ROOT / "RESULTS.md").write_text(document)
    print(json.dumps({"go": results["go"], **analysis}, indent=2))


if __name__ == "__main__":
    main()
